```python
def _create_vector_db(self, embeddings):
    embeddings_array = np.array(embeddings, dtype=np.float32)
    dimension = len(embeddings[0])
    index = faiss.IndexFlatIP(dimension)  # 内积（余弦相似度）
    index.add(embeddings_array)
    return index
```

**索引类型选择**：

### 1.5 LLM 重排序算法 (`reranking.py`)

#### 1.5.1 单块重排

```python
def get_rank_for_single_block(self, query, retrieved_document):
    user_prompt = f'Query: "{query}"\n\nDocument: """{retrieved_document}"""'
    
    # 调用 LLM 评分
    completion = self.llm.beta.chat.completions.parse(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": self.system_prompt_rerank_single_block},
            {"role": "user", "content": user_prompt}
        ],
        response_format=RetrievalRankingSingleBlock
    )
    
    return completion.choices[0].message.parsed
```

#### 1.5.2 批量重排

```python
def rerank_documents(self, query, documents, documents_batch_size=4, llm_weight=0.7):
    # 1. 分批处理
    doc_batches = [documents[i:i+documents_batch_size] 
                   for i in range(0, len(documents), documents_batch_size)]
    
    # 2. 批量调用 LLM
    for batch in doc_batches:
        rankings = self.get_rank_for_multiple_blocks(query, batch)
        for doc, rank in zip(batch, rankings['block_rankings']):
            doc['combined_score'] = (
                llm_weight * rank['relevance_score'] +      # LLM分数
                (1-llm_weight) * doc['distance']             # 向量分数
            )
    
    # 3. 按融合分数排序
    all_results.sort(key=lambda x: x["combined_score"], reverse=True)
    return all_results
```

**融合公式**：
```
combined_score = 0.7 × relevance_score + 0.3 × (1 - distance)
```

```python
def process_comparative_question(self, question, companies, schema):
    # Step 1: 问题重写（LLM）
    rephrased_questions = self.openai_processor.get_rephrased_questions(
        original_question=question,
        companies=companies
    )
    # 输出: {"苹果": "苹果公司2022年营收是多少？", 
    #        "微软": "微软公司2022年营收是多少？"}
    
    # Step 2: 并行处理各公司问题
    with ThreadPoolExecutor() as executor:
        individual_answers = {
            executor.submit(process_company_question, company): company
            for company in companies

        question=question,
        rag_context=individual_answers,
        schema="comparative"
    )
    
    return comparative_answer
```

#### 1.7.3 答案验证

```python
def _validate_page_references(self, claimed_pages, retrieval_results):
    retrieved_pages = [r['page'] for r in retrieval_results]
    
    # 过滤幻觉页码
    validated_pages = [p for p in claimed_pages if p in retrieved_pages]
    
    # 补充至最小页数
    if len(validated_pages) < min_pages:
        for result in retrieval_results:
            if result['page'] not in validated_pages:
                validated_pages.append(result['page'])
                if len(validated_pages) >= min_pages:
                    break
    
    # 截断至最大页数
    return validated_pages[:max_pages]
```

```python
def parse_and_export_parallel(self, input_doc_paths, optimal_workers=10):
    # 计算最优 chunk 大小
    chunk_size = max(1, total_pdfs // optimal_workers)
    
    # 分块分配任务
    chunks = [input_doc_paths[i:i+chunk_size] 
              for i in range(0, total_pdfs, chunk_size)]
    
    with ProcessPoolExecutor(max_workers=optimal_workers) as executor:
        futures = [executor.submit(_process_chunk, chunk, ...) 
                   for chunk in chunks]
```
